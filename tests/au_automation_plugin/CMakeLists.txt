cmake_minimum_required(VERSION 3.18)
project(PedalboardAUAutomationProbe LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "Audio Unit fixtures require macOS")
endif()
set(CMAKE_CXX_STANDARD 17)
set(SDK "${CMAKE_CURRENT_LIST_DIR}/../../JUCE/modules/juce_audio_plugin_client/AU/CoreAudioUtilityClasses")
set(UTILITIES AUBase AUBuffer AUInputElement AUMIDIBase AUOutputElement AUPlugInDispatch
  AUScopeElement CAAudioChannelLayout CAMutex CAStreamBasicDescription CAVectorUnit ComponentBase MusicDeviceBase)
set(SOURCES probe.cpp)
foreach(name IN LISTS UTILITIES)
  list(APPEND SOURCES "${SDK}/${name}.cpp")
endforeach()
add_library(AutomationAU MODULE ${SOURCES})
target_include_directories(AutomationAU PRIVATE "${SDK}")
target_compile_definitions(AutomationAU PRIVATE CA_USE_AUDIO_PLUGIN_ONLY=1 CA_BASIC_AU_FEATURES=0)
target_compile_options(AutomationAU PRIVATE -Wno-deprecated-declarations)
target_link_libraries(AutomationAU PRIVATE "-framework AudioToolbox" "-framework CoreFoundation" "-framework CoreAudio")
set_target_properties(AutomationAU PROPERTIES PREFIX "" SUFFIX ""
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/AutomationAU.component/Contents/MacOS")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/AutomationAU.component/Contents")
configure_file(Info.plist.in "${CMAKE_BINARY_DIR}/AutomationAU.component/Contents/Info.plist" COPYONLY)
