cmake_minimum_required(VERSION 3.18)
project(PedalboardAUv3Probe LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "AUv3 extension fixtures require macOS")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
string(SHA256 PROBE_HASH "${CMAKE_BINARY_DIR}")
string(SUBSTRING "${PROBE_HASH}" 0 12 PROBE_ID)
string(SUBSTRING "${PROBE_HASH}" 0 3 PROBE_PREFIX)
set(APP "${CMAKE_BINARY_DIR}/AUv3 Automation Probe.app")
set(EXT "${APP}/Contents/PlugIns/AutomationAUv3.appex")
file(MAKE_DIRECTORY "${APP}/Contents/MacOS" "${EXT}/Contents/MacOS")
configure_file(AppInfo.plist.in "${APP}/Contents/Info.plist")
configure_file(ExtensionInfo.plist.in "${EXT}/Contents/Info.plist")
add_executable(AutomationProbeHost host.cpp)
set_target_properties(AutomationProbeHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${APP}/Contents/MacOS")
add_executable(AutomationAUv3 AutomationAU.mm)
target_compile_options(AutomationAUv3 PRIVATE -fobjc-arc -fblocks -fapplication-extension)
target_link_options(AutomationAUv3 PRIVATE "-Wl,-e,_NSExtensionMain")
target_link_libraries(AutomationAUv3 PRIVATE "-framework Foundation" "-framework AVFoundation" "-framework AudioToolbox")
set_target_properties(AutomationAUv3 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${EXT}/Contents/MacOS")
